\begin{quote}
{\bfseries{📚 About Doxygen\+:}} Doxygen is a powerful tool that can produce high-\/quality documentation from annotated C/\+C++ sources, making it a critical tool in any developer\textquotesingle{}s toolkit. \end{quote}

\hypertarget{md_docs_2pages_2doxygen__guide_autotoc_md124}{}\doxysubsection{\texorpdfstring{Source Code Browser}{Source Code Browser}}\label{md_docs_2pages_2doxygen__guide_autotoc_md124}

\begin{DoxyCode}{0}
\DoxyCodeLine{\#\ Generate\ source\ browser}
\DoxyCodeLine{SOURCE\_BROWSER\ =\ YES}
\DoxyCodeLine{INLINE\_SOURCES\ =\ YES}
\DoxyCodeLine{STRIP\_CODE\_COMMENTS\ =\ NO}

\end{DoxyCode}


\begin{quote}
{\bfseries{SOURCE\+\_\+\+BROWSER = YES}} ~\newline
 Generates hyperlinked source code ~\newline
 ↓ ~\newline
 {\bfseries{INLINE\+\_\+\+SOURCES = YES}} ~\newline
 Includes source code in documentation ~\newline
 ↓ ~\newline
 {\bfseries{STRIP\+\_\+\+CODE\+\_\+\+COMMENTS = NO}} ~\newline
 Preserves comments in code listings \end{quote}
